Usage

import {
  PHI, KAPPA_STAR, PHI_UNIT,
  evaluateKappa, optimalKappa,
  getTierInfo, deriveByTier,
  success, failure, isSuccess
} from '@konomi/core';

// Get optimal kappa
const kappa = optimalKappa(); // 0.618

// Evaluate kappa state
const state = evaluateKappa(0.6);
// { value: 0.6, label: 'optimal' }

// Get tier info
const t1 = getTierInfo('T1');
// { tier: 'T1', cost: 1.388, cached: false, symbol: '⏱' }

// Result type usage
const result = success({ data: 'value' });
if (isSuccess(result)) {
  console.log(result.value); // { data: 'value' }
}
